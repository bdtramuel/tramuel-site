{{- /* The theme's responsive image, slimmed down so the site stays well under GitHub Pages' 1 GB limit: every size is offered as WebP (which every current browser shows), and only the smallest size is also made as a JPEG for the rare browser that can't. The theme made a JPEG of every size, which roughly doubled the photos' share of the published site. */ -}} {{- $image := .image -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $pictureClass := .pictureClass | default "" -}} {{- $sizes := .sizes | default "100vw" -}} {{- $style := .style | default "" -}} {{- $title := .title | default "" -}} {{- $loading := .loading | default "lazy" -}} {{- $widths := .widths | default (slice 711 1422 2133) -}} {{- if and $image (reflect.IsImageResourceProcessable $image) -}} {{- $fits := slice -}} {{- range $widths -}} {{- if le (int .) $image.Width -}}{{- $fits = $fits | append (int .) -}}{{- end -}} {{- end -}} {{- $webp := slice -}} {{- $fallback := $image -}} {{- if $fits -}} {{- range $fits -}} {{- $webp = $webp | append ($image.Resize (printf "%dx webp Lanczos" .)) -}} {{- end -}} {{- $fallback = $image.Resize (printf "%dx Lanczos" (index $fits 0)) -}} {{- else -}} {{- $webp = $webp | append ($image.Resize (printf "%dx webp Lanczos" $image.Width)) -}} {{- end }} {{- else if and $image (reflect.IsImageResourceWithMeta $image) -}} {{- else if $image -}} {{- end -}}
